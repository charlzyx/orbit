// No window or display is created. This is NOT a native-app capture.
func TestOrbitFontSpecimen(t *testing.T) {
	root := os.Getenv("ORBIT_FONT_TRACE_ASSETS")
	if root == "" {
		t.Skip("set font asset root explicitly")
	}
	e, err := newPango()
	if err != nil {
		t.Fatal(err)
	}
	var paths []string
	err = filepath.WalkDir(root, func(path string, d os.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if !d.IsDir() && (strings.HasSuffix(path, ".ttf") || strings.HasSuffix(path, ".otf")) {
			paths = append(paths, path)
		}
		return nil
	})
	if err != nil {
		t.Fatal(err)
	}
	sort.Strings(paths) // same lexical asset-walk ordering as Go LoadAssets
	for _, path := range paths {
		b, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		family := ""
		if strings.Contains(strings.ToLower(path), "plexsans") {
			family = ".ZedSans"
		}
		if strings.Contains(strings.ToLower(path), "lilex") {
			family = ".ZedMono"
		}
		if err = e.register(b, family); err != nil {
			t.Fatal(err)
		}
	}
	s := newSystem()
	s.eng = e
	if os.Getenv("ORBIT_COMPAT_SPECIMEN_ENABLE") == "1" {
		if err := s.EnableOrbitFractionalTextMetrics(); err != nil {
			t.Fatal(err)
		}
	}
	var out []orbitEvent
	for _, family := range []string{".ZedSans", ".ZedMono"} {
		for _, italic := range []bool{false, true} {
			for _, weight := range []int{400, 500, 600, 700} {
				for _, value := range []string{"General", "This machine", "AV office fi 0123456789 → 中文 🙂"} {
					p := Params{Text: value, Style: Style{Family: family, Size: 14, Weight: weight, Italic: italic, LineHeight: 1.3, Lang: "en"}}
					l := s.Shape(p)
					var lines []orbitEvent
					for _, line := range l.Lines {
						var glyphs []orbitEvent
						for _, g := range line.Glyphs {
							glyphs = append(glyphs, orbitEvent{"id": g.ID, "x": g.X, "y": g.Y, "advance": g.Advance, "cluster": g.Cluster, "ascent": g.Font.Ascent, "descent": g.Font.Descent})
						}
						lines = append(lines, orbitEvent{"width": line.Width, "height": line.Height, "ascent": line.Ascent, "descent": line.Descent, "baseline": line.Baseline, "glyphs": glyphs})
					}
					out = append(out, orbitEvent{"text": value, "style": p.Style, "lines": lines})
				}
			}
		}
	}
	if path := os.Getenv("ORBIT_FONT_SPECIMEN_RESULT"); path != "" {
		b, err := json.MarshalIndent(out, "", "  ")
		if err != nil {
			t.Fatal(err)
		}
		if err = os.WriteFile(path, b, 0600); err != nil {
			t.Fatal(err)
		}
	}
	orbitFlush("headless native text-engine specimen; NO window, NO native present")
	if st := orbitState(); st != nil { // Test only: ensure writer has finished before process exits.
		deadline := time.Now().Add(5 * time.Second)
		for time.Now().Before(deadline) {
			files, _ := filepath.Glob(filepath.Join(st.dir, "mygo-*.json"))
			if len(files) > 0 {
				if b, err := os.ReadFile(files[0]); err == nil && json.Valid(b) {
					return
				}
			}
			time.Sleep(10 * time.Millisecond)
		}
		t.Fatal("trace batch did not flush")
	}
}

// Exercise the same getter-only observation path used by actual shape events.
func TestOrbitObservedContextPolicy(t *testing.T) {
	for _, fractional := range []bool{false, true} {
		s := orbitMetricSystem(t)
		if fractional {
			if err := s.EnableOrbitFractionalTextMetrics(); err != nil {
				t.Fatal(err)
			}
		}
		e := s.eng.(*pangoEngine)
		wantMetrics := int32(cairoHintMetricsOn)
		if fractional {
			wantMetrics = cairoHintMetricsOff
		}
		for _, hint := range []string{"medium", "slight", "full", "medium"} {
			s.SetFontRendering("gray", hint, "rgb")
			rec := e.orbitContextPolicy()
			if rec["round_positions"] != !fractional || rec["context_hint_metrics"] != wantMetrics {
				t.Fatalf("fractional=%v hint=%s actual policy=%v", fractional, hint, rec)
			}
			if rec["context_antialias"] != int32(cairoAntialiasGray) {
				t.Fatalf("actual antialias=%v", rec)
			}
		}
		// Test-only native mutation proves the observer reads actual state,
		// not the candidate's intended policy field or setter availability.
		pangoLib.contextSetRoundPositions(e.context, 1)
		if e.orbitContextPolicy()["round_positions"] != true {
			t.Fatal("observer inferred intended rounding")
		}
		e.applyOptions()
	}
}
