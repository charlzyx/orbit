//go:build linux && (amd64 || arm64)

package text

import (
	"encoding/json"
	"os"
	"path/filepath"
	"sort"
	"strings"
	"testing"
)

// Diagnostic fixture only. No window, display, observer or native-app paint.
// The environment selects the TEST call; production has no environment switch.
func TestOrbitTextMetricsSpecimen(t *testing.T) {
	root := os.Getenv("ORBIT_FONT_TRACE_ASSETS")
	if root == "" {
		t.Skip("set font asset root explicitly")
	}
	e, err := newPango()
	if err != nil {
		t.Fatal(err)
	}
	var paths []string
	err = filepath.WalkDir(root, func(path string, d os.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if !d.IsDir() && (strings.HasSuffix(path, ".ttf") || strings.HasSuffix(path, ".otf")) {
			paths = append(paths, path)
		}
		return nil
	})
	if err != nil {
		t.Fatal(err)
	}
	sort.Strings(paths) // same lexical asset-walk ordering as Go LoadAssets
	for _, path := range paths {
		b, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		family := ""
		if strings.Contains(strings.ToLower(path), "plexsans") {
			family = ".ZedSans"
		}
		if strings.Contains(strings.ToLower(path), "lilex") {
			family = ".ZedMono"
		}
		if err = e.register(b, family); err != nil {
			t.Fatal(err)
		}
	}
	s := newSystem()
	s.eng = e
	if os.Getenv("ORBIT_COMPAT_SPECIMEN_ENABLE") == "1" {
		if err := s.EnableOrbitFractionalTextMetrics(); err != nil {
			t.Fatal(err)
		}
	}
	type orbitEvent = map[string]any
	var out []orbitEvent
	for _, family := range []string{".ZedSans", ".ZedMono"} {
		for _, italic := range []bool{false, true} {
			for _, weight := range []int{400, 500, 600, 700} {
				for _, value := range []string{"General", "This machine", "AV office fi 0123456789 → 中文 🙂"} {
					p := Params{Text: value, Style: Style{Family: family, Size: 14, Weight: weight, Italic: italic, LineHeight: 1.3, Lang: "en"}}
					l := s.Shape(p)
					var lines []orbitEvent
					for _, line := range l.Lines {
						var glyphs []orbitEvent
						for _, g := range line.Glyphs {
							glyphs = append(glyphs, orbitEvent{"id": g.ID, "x": g.X, "y": g.Y, "advance": g.Advance, "cluster": g.Cluster, "ascent": g.Font.Ascent, "descent": g.Font.Descent})
						}
						lines = append(lines, orbitEvent{"width": line.Width, "height": line.Height, "ascent": line.Ascent, "descent": line.Descent, "baseline": line.Baseline, "glyphs": glyphs})
					}
					out = append(out, orbitEvent{"text": value, "style": p.Style, "lines": lines})
				}
			}
		}
	}
	if path := os.Getenv("ORBIT_FONT_SPECIMEN_RESULT"); path != "" {
		b, err := json.MarshalIndent(out, "", "  ")
		if err != nil {
			t.Fatal(err)
		}
		if err = os.WriteFile(path, b, 0600); err != nil {
			t.Fatal(err)
		}
	}
}
